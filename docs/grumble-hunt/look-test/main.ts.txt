const ramp = new T.DataTexture(new Uint8Array([88, 165, 255]), 3, 1, T.RedFormat);
ramp.minFilter = ramp.magFilter = T.NearestFilter; ramp.generateMipmaps = false; ramp.needsUpdate = true;
type Pair = { soft: T.Material; toon: T.Material };
const swaps: T.Mesh[] = [];
function pair(color: number, o: { rough?: number; vc?: boolean; outline?: boolean; metal?: number; map?: T.Texture } = {}): Pair {
  const soft = new T.MeshStandardMaterial({ color, roughness: o.rough ?? 0.86, metalness: o.metal ?? 0, vertexColors: !!o.vc, map: o.map ?? null });
  const toon = new T.MeshToonMaterial({ color, gradientMap: ramp, vertexColors: !!o.vc, map: o.map ?? null });
  if (o.outline === false) toon.userData.outlineParameters = { visible: false };
  return { soft, toon };
}
function mesh(geo: T.BufferGeometry, p: Pair, parent: T.Object3D, cast = true) {
  const m = new T.Mesh(geo, p.soft); m.userData.pair = p; m.castShadow = cast; m.receiveShadow = true; parent.add(m); swaps.push(m); return m;
}
function windify(m: T.Material, amp: number) {
  m.onBeforeCompile = (sh) => {
    sh.uniforms.uTime = timeU;
    sh.vertexShader = 'uniform float uTime;\n' + sh.vertexShader.replace('#include <begin_vertex>', `#include <begin_vertex>
      vec4 ip = instanceMatrix[3];
      float w = sin(uTime*1.6 + ip.x*.31 + ip.z*.23)*.6 + sin(uTime*3.3 + ip.x*.8 - ip.z*.5)*.2;
      float h = position.y;
      transformed.x += w*h*h*${amp.toFixed(2)}; transformed.z += w*h*h*${(amp * 0.5).toFixed(2)};`);
  };
}

const toonCache = new Map<T.Material, T.Material>();
function adopt(root: T.Object3D) {
  root.traverse((o) => {
    if (!(o instanceof T.Mesh)) return;
    const m = o.material as T.Material;
    let t = toonCache.get(m);
    if (!t) { t = m instanceof T.MeshStandardMaterial ? new T.MeshToonMaterial({ color: m.color.clone(), gradientMap: ramp }) : m; toonCache.set(m, t); }
    o.userData.pair = { soft: m, toon: t }; swaps.push(o);
  });
}
function rig(white: boolean) {
  const model = createHuntChinchilla(white); adopt(model.root);
  const box = new T.Box3().setFromObject(model.root), s = 1.0 / (box.max.y - box.min.y);
  model.root.scale.setScalar(s);
  const g = new T.Group(); g.add(model.root); scene.add(g);
  return { g, model, s };
}

function stepParts(dt: number) {
  for (let i = parts.length - 1; i >= 0; i--) {
    const p = parts[i]; p.life += dt; const f = p.life / p.max;
    if (f >= 1) { scene.remove(p.m); if (p.m instanceof T.Mesh && p.kind === 'dust') swaps.splice(swaps.indexOf(p.m), 1); if (p.m instanceof T.Sprite) p.m.material.dispose(); parts.splice(i, 1); continue; }
    p.m.position.addScaledVector(p.vel, dt);
    if (p.kind === 'spark') { p.vel.y -= 14 * dt; p.m.scale.setScalar(p.size * 0.35 * (1 - f)); p.m.rotation.x += dt * 9; }
    else if (p.kind === 'dust') { p.vel.multiplyScalar(1 - 3 * dt); p.m.scale.setScalar(p.size * Math.sin(Math.min(1, f * 1.4) * Math.PI) * (1 + f)); }
    else { p.m.scale.setScalar(0.5 + f * 0.4); (p.m as T.Sprite).material.opacity = 1 - f; }
  }
}

    if (p.t > 1) { p.el.remove(); pops.splice(i, 1); }
  }
  frames++; fpsT += real; if (fpsT > 0.5) { hud.fps.textContent = `${Math.round(frames / fpsT)} fps`; frames = 0; fpsT = 0; }
  requestAnimationFrame(frame);
}
